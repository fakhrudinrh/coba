$.fn.dataTable.ext.errMode = 'throw';
$('#@Model.TableId').on('error.dt', function(e, settings, techNote, message){
    var columnCount = settings.aoColumns.length;;
    var errorRow = `<tr class="error-row"><td colspan="${columnCount}" style="text-align: center; color: red; background-color: #fff0f0;">
        <strong>An error occurred:</strong> Failed to load data from the server.</td></tr>`;

    $(this).find('tbody').html(errorRow);

    if (settings.oFeatures.bProcessing) {
        $(`#${settings.sTableId}_processing`).hide();
    }
});
